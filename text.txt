    function stopMusic() {
      if (!audio.ready) return;
      if (audio.musicInterval) clearInterval(audio.musicInterval);
      if (audio.melodyInterval) clearInterval(audio.melodyInterval);
      audio.musicInterval = null;
      audio.melodyInterval = null;
      audio.currentTheme = -1;
      audio.musicGain.gain.setTargetAtTime(0.00001, audio.ctx.currentTime, 0.04);
    }

    function applyLevelMusic(level) {
      if (!audio.ready) return;
      const themeIndex = (level - 1) % musicThemes.length;
      if (audio.currentTheme === themeIndex) return;
      audio.currentTheme = themeIndex;
      audio.musicGain.gain.setTargetAtTime(0.022, audio.ctx.currentTime, 0.08);
      if (audio.musicInterval) clearInterval(audio.musicInterval);
      if (audio.melodyInterval) clearInterval(audio.melodyInterval);
      const data = musicThemes[themeIndex];
      let chordIndex = 0;
      let melodyIndex = 0;
      audio.musicInterval = setInterval(() => {
        const chord = data.chords[chordIndex % data.chords.length];
        chordIndex++;
        audio.bassOsc.frequency.setTargetAtTime(chord[0], audio.ctx.currentTime, 0.5);
        audio.padOsc1.frequency.setTargetAtTime(chord[1], audio.ctx.currentTime, 0.5);
        audio.padOsc2.frequency.setTargetAtTime(chord[2], audio.ctx.currentTime, 0.5);
      }, 2400);
      function tickMelody() {
        if (audio.currentTheme !== themeIndex) return;
        const note = data.melody[melodyIndex % data.melody.length];
        melodyIndex++;
        audio.melodyOsc.frequency.setTargetAtTime(note, audio.ctx.currentTime, 0.08);
        audio.melodyInterval = setTimeout(tickMelody, zoomLocked ? data.speed * 0.48 : data.speed);
      }
      tickMelody();
    }

    function beep(freq, duration, type, volume) {
      if (!audio.ready) return;
      const osc = audio.ctx.createOscillator();
      const gain = audio.ctx.createGain();
      osc.type = type || "sine";
      osc.frequency.value = freq || 440;
      gain.gain.value = volume || 0.06;
      osc.connect(gain);
      gain.connect(audio.ctx.destination);
      osc.start();
      gain.gain.exponentialRampToValueAtTime(0.001, audio.ctx.currentTime + (duration || 0.1));
      osc.stop(audio.ctx.currentTime + (duration || 0.1));
    }

    function formatTime(ms) {
      const totalSeconds = Math.max(0, Math.floor(ms / 1000));
      const minutes = Math.floor(totalSeconds / 60);

      return `${minutes}:${String(seconds).padStart(2, "0")}`;
    }

    async function getLeaderboard() {
      if (GLOBAL_LEADERBOARD_ENABLED && globalDb) {
        const snapshot = await globalDb
          .collection("leaderboard")
          .orderBy("time", "asc")
          .limit(20)
          .get();

        return snapshot.docs.map(doc => doc.data());
      }

      try {
        return JSON.parse(localStorage.getItem(LEADERBOARD_KEY) || "[]");
      } catch {
        return [];
      }
    }

    }

    async function saveScoreOnlineOrLocal(username, time) {
      if (GLOBAL_LEADERBOARD_ENABLED && globalDb) {
        await globalDb.collection("leaderboard").add({
          username,
          usernameLower: username.toLowerCase(),
          time,
          date: Date.now()
        });
        return;
      }

      const entries = await getLeaderboard();
      entries.push({ username, usernameLower: username.toLowerCase(), time, date: Date.now() });
      entries.sort((a, b) => a.time - b.time);
      saveLeaderboard(entries);
    }

    async function renderLeaderboard() {
      const entries = (await getLeaderboard()).sort((a, b) => a.time - b.time);
      leaderboardList.innerHTML = "";

      if (entries.length === 0) {
        const empty = document.createElement("li");
        empty.textContent = "No completions yet. Be the first Rudi legend.";
        leaderboardList.appendChild(empty);
        return;
      }

      for (const entry of entries) {
        const item = document.createElement("li");
        item.textContent = `${entry.username} — ${formatTime(entry.time)}`;
        leaderboardList.appendChild(item);
      }
    }

    function updateJoystick(e) {
      const rect = joystickBase.getBoundingClientRect();
      const cx = rect.left + rect.width / 2;
      const cy = rect.top + rect.height / 2;
      let dx = e.clientX - cx;
      let dy = e.clientY - cy;
      const dist = Math.hypot(dx, dy);
      const max = 42;
      if (dist > max) {
        dx = dx / dist * max;
        dy = dy / dist * max;
      }
      joy.x = dx / max;
      joy.y = dy / max;
      joystickKnob.style.left = `${36 + dx}px`;
      joystickKnob.style.top = `${36 + dy}px`;
    }

    function easeOutBack(t) {
      const c1 = 1.70158;
      const c3 = c1 + 1;
      return 1 + c3 * Math.pow(t - 1, 3) + c1 * Math.pow(t - 1, 2);
    }

    function easeInOut(t) {
      return t < 0.5 ? 2 * t * t : 1 - Math.pow(-2 * t + 2, 2) / 2;
    }

    function getFormScale(form) {
      if (form === "big") return new THREE.Vector3(2.35, 2.7, 2.35);
      if (form === "small") return new THREE.Vector3(0.48, 0.62, 0.48);
      if (form === "speed") return new THREE.Vector3(1.22, 1.18, 1.55);
      if (form === "clone") return new THREE.Vector3(0.9, 1.7, 0.9);
      return new THREE.Vector3(1.08, 1.38, 1.08);
    }

    function animateRudiPower(nextForm) {
      const fromForm = currentPowerForm;
      const baseFrom = rudi.scale.clone();
      const baseTo = getFormScale(nextForm);
      let midScale = null;
      let duration = 0.45;
      let style = "normal";

      if (nextForm === "big") {
        midScale = baseTo.clone().multiply(new THREE.Vector3(1.12, 0.86, 1.12));
        duration = 0.55;
        style = "growSquash";
      }

      if (nextForm === "small") {
        duration = fromForm === "big" ? 0.72 : 0.5;
        style = "shrink";
      }

      if (nextForm === "speed") {
        midScale = new THREE.Vector3(0.92, 1.55, 1.85);
        duration = 0.42;
        style = "stretchZoom";
      }

      if (nextForm === "clone") {
        midScale = baseFrom.clone().multiply(new THREE.Vector3(0.72, 1.45, 1.2));
        duration = 0.5;
        style = "clonePop";
      }

      if (nextForm === "normal") {
        duration = 0.5;
        style = "settle";
      }

      powerAnim = {
        time: 0,
        duration,
        from: baseFrom,
        mid: midScale,
        to: baseTo,
        style
      };

      currentPowerForm = nextForm;
    }

    function updatePowerAnimation(dt) {
      if (!powerAnim) return;

      powerAnim.time += dt;
      const t = Math.min(1, powerAnim.time / powerAnim.duration);

      let target;

      if (powerAnim.mid) {
        if (t < 0.5) {
          target = powerAnim.from.clone().lerp(powerAnim.mid, easeOutBack(t * 2));
        } else {
          target = powerAnim.mid.clone().lerp(powerAnim.to, easeOutBack((t - 0.5) * 2));
        }
      } else {
        target = powerAnim.from.clone().lerp(powerAnim.to, easeInOut(t));
      }

      if (powerAnim.style === "shrink") {
        rudi.rotation.z += Math.sin(t * Math.PI) * 0.025;
      }

      if (powerAnim.style === "clonePop") {
        rudi.position.y += Math.sin(t * Math.PI) * 0.35;
      }

      rudi.scale.copy(target);

      if (t >= 1) {
        rudi.scale.copy(powerAnim.to);
        powerAnim = null;
      }
    }

          hint.textContent = "THE CLONE ACHIEVED MAXIMUM ZOOMIES";
        }
      }
      if (type === "big") {
        sizeMultiplier = 2.2;
        animateRudiPower("big");
        setTimeout(() => {
          sizeMultiplier = 1;
          animateRudiPower("normal");
        }, 12000);
      }
      if (type === "small") {
        sizeMultiplier = 0.45;
        animateRudiPower("small");
        setTimeout(() => {
          sizeMultiplier = 1;
          animateRudiPower("normal");
        }, 12000);
      }
      if (type === "clone3") {
        animateRudiPower("clone");
        createClone();
        createClone();
        createClone();
        setTimeout(() => animateRudiPower("normal"), 520);
      }
      if (type === "clone1") {
        animateRudiPower("clone");
        createClone();
        setTimeout(() => animateRudiPower("normal"), 520);
      }
    }

    function addDebris(position, material, count, force) {
      for (let i = 0; i < count; i++) {
        const debris = new THREE.Mesh(
          new THREE.BoxGeometry(Math.random() * 3 + 1, Math.random() * 3 + 1, Math.random() * 3 + 1),
          material ? material.clone() : mat([0x999999, 0x777777, 0xbbbbbb][Math.floor(Math.random() * 3)])
        );
        debris.position.copy(position);
        const angle = Math.random() * Math.PI * 2;
        const launchForce = Math.random() * force + 5;
        debris.userData.vel = new THREE.Vector3(
          Math.cos(angle) * launchForce,
          Math.random() * 11 + 4,
          Math.sin(angle) * launchForce
        );
        debris.userData.spin = new THREE.Vector3(
          Math.random() * 0.25,
          Math.random() * 0.25,
          Math.random() * 0.25
        );
        debris.castShadow = true;
        debris.receiveShadow = true;
        scene.add(debris);
        debrisObjects.push(debris);
      }
    }

    function explodeRoom() {
      explosionFlash = 1;
      explosionOverlay.style.opacity = "1";

          breakables.push(obj);

      for (const obj of breakables) {

        obj.visible = false;

      for (let i = 0; i < 90; i++) {
        const angle = Math.random() * Math.PI * 2;
        const radius = Math.random() * 55;
        addDebris(
          new THREE.Vector3(Math.cos(angle) * radius, Math.random() * 18 + 2, Math.sin(angle) * radius),
          null,
          1,
          12
        );
      }

      beep(18, 2.8, "sawtooth", 0.24);
    }

    function updateGameplay(dt) {
      let mx = 0;
      let mz = 0;

      if (keys.w || keys.arrowup) mz += 1;
      if (keys.s || keys.arrowdown) mz -= 1;
      if (keys.a || keys.arrowleft) mx -= 1;
      if (keys.d || keys.arrowright) mx += 1;

      mx += joy.x;
      mz += -joy.y;

      const forwardX = Math.sin(cameraYaw);
      const forwardZ = Math.cos(cameraYaw);
      const rightX = -Math.cos(cameraYaw);
      const rightZ = Math.sin(cameraYaw);
      const worldMX = rightX * mx + forwardX * mz;
      const worldMZ = rightZ * mx + forwardZ * mz;
      const inputLen = Math.hypot(worldMX, worldMZ);
      const zooming = zoomLocked && boost > 0;

      if (audio.ready && audio.musicGain) {
        audio.musicGain.gain.setTargetAtTime(zooming ? 0.035 : 0.022, audio.ctx.currentTime, 0.08);
      }

      const topSpeed = (zooming ? 24 : 9) * speedMultiplier;

      if (inputLen > 0.05) {
        const moveX = worldMX / inputLen;
        const moveZ = worldMZ / inputLen;
        velocity.x += moveX * topSpeed * dt * 6;
        velocity.z += moveZ * topSpeed * dt * 6;
        if (!rudySpinMode) rudi.rotation.y = Math.atan2(moveX, moveZ) - Math.PI / 2;
      }

      velocity.multiplyScalar(0.88);
      if (velocity.length() > topSpeed) velocity.setLength(topSpeed);
      rudiPos.addScaledVector(velocity, dt);
      rudiPos.x = THREE.MathUtils.clamp(rudiPos.x, -720, 720);
      rudiPos.z = THREE.MathUtils.clamp(rudiPos.z, -720, 720);

      rudi.position.copy(rudiPos);
      if (!powerAnim) {
        if (currentPowerForm === "normal") {
          rudi.scale.set(1.08 * sizeMultiplier, 1.38 * sizeMultiplier, 1.08 * sizeMultiplier);
        }
      }
      updatePowerAnimation(dt);

      const moving = velocity.length() > 0.35;
      const animTime = performance.now() * 0.012;
      const zoomBonus = zooming ? 1.7 : 1;
      rudi.position.y = moving ? Math.abs(Math.sin(animTime * zoomBonus)) * 0.22 : Math.sin(animTime * 0.35) * 0.035;

      for (let i = 0; i < rudiParts.legs.length; i++) {
        rudiParts.legs[i].rotation.z = moving ? Math.sin(animTime * 1.8 * zoomBonus + (i % 2 === 0 ? 0 : Math.PI)) * 0.45 : 0;
      }

      if (rudiParts.tail) rudiParts.tail.rotation.y = Math.sin(animTime * 2.7 * zoomBonus) * 0.75;

      for (let i = 0; i < rudiParts.ears.length; i++) {
        rudiParts.ears[i].rotation.x = (i === 0 ? 0.15 : -0.15) + (moving ? Math.sin(animTime * 1.5 + i) * 0.18 : 0);
      }

      if (rudiParts.head) rudiParts.head.rotation.z = moving ? Math.sin(animTime * 0.9) * 0.08 : 0;
      if (rudySpinMode) rudi.rotation.z += dt * 5;
      if (rudyDanceMode && rudiParts.head) rudiParts.head.rotation.z = Math.sin(performance.now() * 0.014) * 0.25;

      if (zooming) boost = Math.max(0, boost - dt * 22);
      else boost = Math.min(100, boost + dt * 18);
      boostEl.textContent = Math.round(boost);

      endingStar.rotation.y += dt * 2;
      endingStar.rotation.x += dt * 1.5;
      if (endingStar.parent && endingStar.position.distanceTo(rudiPos) < 7) {
        scene.remove(endingStar);
        triggerMeteorEnding();
      }

      for (const treat of treats) {
        treat.rotation.z += dt * 2.5;
        treat.rotation.y += dt * 1.2;
        if (treat.position.distanceTo(rudiPos) < 3.1) collectTreat(treat, false);
      }

      for (let i = powerUps.length - 1; i >= 0; i--) {
        const power = powerUps[i];
        power.rotation.y += dt * 2.5;
        power.rotation.x += dt * 1.6;

        if (power.userData.type === "clone3") power.material.color.setHSL((performance.now() * 0.00025) % 1, 1, 0.6);
        if (power.userData.type === "clone1") {
          const pulse = (Math.sin(performance.now() * 0.018) + 1) / 2;
          power.material.color.setRGB(0, 0.55 + pulse * 0.45, 1);
        }

        if (power.position.distanceTo(rudiPos) < 5) {
          applyPowerUp(power.userData.type);
          scene.remove(power);
          powerUps.splice(i, 1);
          powerCountEl.textContent = powerUps.length;
        }
      }

      updateClones(dt);

      const camDistance = 20;
      const safePitch = Math.max(cameraPitch, -1.05);
      const horizontalDistance = Math.cos(safePitch) * camDistance;
      const verticalOffset = Math.sin(safePitch) * camDistance + 10;
      const camX = rudiPos.x - Math.sin(cameraYaw) * horizontalDistance;
      const camZ = rudiPos.z - Math.cos(cameraYaw) * horizontalDistance;
      const camY = Math.max(2.5, verticalOffset);
      camera.position.lerp(new THREE.Vector3(camX, camY, camZ), 0.11);
      camera.lookAt(rudiPos.x, 1.6, rudiPos.z);
    }

    function updateClones(dt) {
      for (let i = 0; i < clones.length; i++) {
        const clone = clones[i];

        if (clone.userData.flyAway) {
          clone.position.addScaledVector(clone.userData.flyVelocity, dt);
          clone.userData.flyVelocity.y += dt * 8;
          continue;
        }

        const angle = ((Math.PI * 2) / Math.max(1, clones.length)) * i + performance.now() * 0.00015;
        const radius = 8 + clones.length * 1.8;
        const orbitTarget = new THREE.Vector3(
          rudi.position.x + Math.sin(angle) * radius,
          rudi.position.y,
          rudi.position.z + Math.cos(angle) * radius
        );

        clone.position.lerp(orbitTarget, 0.06);
        clone.lookAt(rudi.position.x, clone.position.y, rudi.position.z);

        for (const treat of treats) {
          if (treat.position.distanceTo(clone.position) < 3.1) collectTreat(treat, true);
        }

        for (let p = powerUps.length - 1; p >= 0; p--) {
          const power = powerUps[p];
          if (power.position.distanceTo(clone.position) < 5) {
            applyPowerUp(power.userData.type);
            scene.remove(power);
            powerUps.splice(p, 1);
            powerCountEl.textContent = powerUps.length;
          }
        }
      }
    }

    function updateDebris() {
      for (const debris of debrisObjects) {
        debris.position.add(debris.userData.vel);
        debris.userData.vel.y -= 0.08;
        debris.rotation.x += debris.userData.spin.x;
        debris.rotation.y += debris.userData.spin.y;
        debris.rotation.z += debris.userData.spin.z;
      }
    }

    function resetGameToMenu() {
      flyAwayEnding = false;
      flyAwayTimer = 0;
      meteorEnding = false;
      meteorTimer = 0;
      meteorExploded = false;
      explosionFlash = 0;
      rudySpinMode = false;
      rudyDanceMode = false;
      cloneFlyAwayTriggered = false;
      score = 0;
      boost = 100;
      speedStacks = 0;
      speedMultiplier = 1;
      sizeMultiplier = 1;
      zoomLocked = false;
      zoomResetArmed = false;
      cloneClearArmed = false;
      scoreEl.textContent = "0";
      boostEl.textContent = "100";
      zoomStacksButton.style.background = "rgba(255,255,255,.82)";
      cloneCountButton.style.background = "rgba(255,255,255,.82)";
      updateCounters();

      endText.style.display = "none";
      titleScreen.classList.remove("hidden");
      hud.style.display = "block";
      hint.style.display = "block";
      joystickBase.style.display = "block";
      zoomButton.style.display = "block";
      zoomButton.style.transform = "scale(1)";
      zoomButton.style.background = "rgba(255,255,255,.84)";
      explosionOverlay.style.opacity = "0";
      stopMusic();

      for (const debris of debrisObjects) scene.remove(debris);
      debrisObjects.length = 0;
      while (clones.length) scene.remove(clones.pop());
      updateCounters();

      rudi.visible = true;
      rudiPos.set(0, 0, 0);
      rudi.position.copy(rudiPos);
      rudi.rotation.set(0, 0, 0);
      rudi.scale.set(1.08, 1.38, 1.08);
      powerAnim = null;
      currentPowerForm = "normal";
      velocity.set(0, 0, 0);

      meteor.visible = false;
      meteorLight.intensity = 0;
      if (!endingStar.parent) scene.add(endingStar);
      moveEndingStar();

      cameraYaw = 0;
      cameraPitch = 0.35;
      camera.position.set(0, 14, 19);
      camera.lookAt(0, 1.6, 0);
      gameStarted = false;
    }

    document.getElementById("goBackButton").addEventListener("pointerdown", e => {
      e.preventDefault();
      resetGameToMenu();
    });

    zoomStacksButton.addEventListener("pointerdown", e => {
      e.preventDefault();
      e.stopPropagation();

      if (!zoomResetArmed) {
        zoomResetArmed = true;
        zoomStacksButton.style.background = "#ffb3b3";
        hint.textContent = "Tap Zoom stacks again to reset all speed stacks";
        return;
      }

      speedStacks = 0;
      speedMultiplier = 1;
      zoomResetArmed = false;
      zoomStacksButton.style.background = "rgba(255,255,255,.82)";
      updateCounters();
      hint.textContent = "Zoom stacks reset. Rudi is normal speed again.";
    });

    cloneCountButton.addEventListener("pointerdown", e => {
      e.preventDefault();
      e.stopPropagation();

      if (!cloneClearArmed) {
        cloneClearArmed = true;
        cloneCountButton.style.background = "#ffb3b3";
        hint.textContent = "Tap Clones again to remove all clones";
        return;
      }

      while (clones.length) scene.remove(clones.pop());
      cloneClearArmed = false;
      cloneCountButton.style.background = "rgba(255,255,255,.82)";
      updateCounters();
      hint.textContent = "All clones removed.";
    });

    spawnLevel(currentLevel);
    updateCounters();
    stopMusic();

    const clock = new THREE.Clock();

    function animate() {
      requestAnimationFrame(animate);
      const dt = Math.min(clock.getDelta(), 0.033);

      if (!gameStarted) {
        updateTitle(dt);
        return;
      }

      if (flyAwayEnding) updateFlyAwayEnding(dt);
      else if (meteorEnding) updateMeteorEnding(dt);
      else updateGameplay(dt);

      updateDebris();
      renderer.render(scene, camera);
    }

    animate();

    window.addEventListener("resize", () => {
      camera.aspect = window.innerWidth / window.innerHeight;
      camera.updateProjectionMatrix();
      renderer.setSize(window.innerWidth, window.innerHeight);
      titleRenderer.setSize(titleBox.clientWidth || 320, titleBox.clientHeight || 260);
    });
  </script>
